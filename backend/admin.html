pkg update -y && pkg install git curl tar -y

# 2. Direct AI Studio / VPS se app source download karein
curl -L -o telebot.tar.gz "/api/download/flutter-app"

# 3. Code extract karein
tar -xzf telebot.tar.gz

# 4. Apne naye GitHub repository par push karein

git branch -M main
git remote add origin https://github.com/APNA_GITHUB_USERNAME/telebot-app.git
